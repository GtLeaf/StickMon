cmake_minimum_required(VERSION 3.16)
set(IDF_TARGET esp32s3)
set(STICKMON_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(stickmon_ostb_3st)
idf_build_set_property(CXX_COMPILE_OPTIONS "-std=gnu++17" APPEND)
set(RESOURCE_STAGE_DIR "${CMAKE_BINARY_DIR}/stickmon_resources")
file(REMOVE_RECURSE "${RESOURCE_STAGE_DIR}")
file(MAKE_DIRECTORY "${RESOURCE_STAGE_DIR}")
file(GLOB_RECURSE RESOURCE_FILES CONFIGURE_DEPENDS "${STICKMON_ROOT}/data/*")
set(RESOURCE_DEPENDS)
foreach(RESOURCE IN LISTS RESOURCE_FILES)
    if(RESOURCE MATCHES "\\.(json|smonsp|smonroom|smonfurn|smonfx|smonfont|smonaudio|smoncry)$" AND NOT RESOURCE MATCHES "ui_amoled")
        file(RELATIVE_PATH RELATIVE "${STICKMON_ROOT}/data" "${RESOURCE}")
        get_filename_component(PARENT "${RESOURCE_STAGE_DIR}/${RELATIVE}" DIRECTORY)
        file(MAKE_DIRECTORY "${PARENT}")
        configure_file("${RESOURCE}" "${RESOURCE_STAGE_DIR}/${RELATIVE}" COPYONLY)
        list(APPEND RESOURCE_DEPENDS "${RESOURCE}")
    endif()
endforeach()
spiffs_create_partition_image(resources "${RESOURCE_STAGE_DIR}" FLASH_IN_PROJECT DEPENDS ${RESOURCE_DEPENDS})
